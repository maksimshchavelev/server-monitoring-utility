52 template <typename ModuleType>
53 void register_module()
54 requires std::is_base_of_v<IModule, ModuleType>
55 {
56 static_assert(
57 requires { ModuleType(Json::Value()); },
58 "Module must be constructible from const Json::Value&");
59
60
61 // Print log
62 logger().log_colorless(
63 std::format("Registering a module with name "
64 "\"\033[36m{}\033[0m\" and description \"\033[36m{}\033[0m\"...",
65 ModuleType::module_name_static(),
66 ModuleType::module_description_static()));
67
68 // Getting config
69 auto config = Config_IO::instance().get_module_config(ModuleType::module_name_static());
70
71 // For example, [MODULE RAM]
72 const std::string module_log_prefix =
73 std::format("[MODULE \033[36m{}\033[0m] ", ModuleType::module_name_static());
74
75 if (config.empty()) {
76 logger().log_colorless(
77 std::format("{}\033[33mGot empty config. Continuing with default values\033[30m",
78 module_log_prefix));
79 }
80
81
82 // Creating module
83 std::unique_ptr<ModuleType> module;
84 bool creation_failed{false};
85
86 try {
87 module = std::make_unique<ModuleType>(std::move(config));
88 // Force first data request. Otherwise, the core will wait until the counter reaches
89 // `poll_ratio - 1`.
90 module->m_poll_counter = module->get_poll_ratio();
91 } catch (const std::exception& e) {
92 creation_failed = true;
93 logger().log_colorless(std::format(
94 "{}\033[31mRegistration failed! Cause: {}\033[30m\n", module_log_prefix, e.what()));
95 }
96
97 if (!creation_failed) {
98 // Get colorful status (RUNNING/STOPPED)
99 std::string module_status =
100 module->is_enabled() ? "\033[32mRUNNING\033[0m" : "\033[31mSTOPPED\033[0m";
101
102 // Print colorful log
103 logger().log_colorless(std::format(
104 "{}\033[32mRegistered\033[0m ({})\n", module_log_prefix, module_status));
105
106 std::lock_guard<std::mutex> lock(m_modules_mutex);
107 m_modules.push_back(std::move(module));
108 }
109 }
