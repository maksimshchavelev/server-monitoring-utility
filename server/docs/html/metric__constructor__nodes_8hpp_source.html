308template <typename... Children> class MetricContainerNode : public IMetricNode<Children...> {
309 public:
321 MetricContainerNode(const std::string& node_name,
322 bool is_root_container,
323 Children&&... children) :
324 IMetricNode<Children...>(
325 node_name, "", "", is_root_container, std::forward<Children>(children)...) {}
326

329
347 Json::Value to_json() const override {
348 Json::Value root;
349
350 // We don't need to store the type if we are root node
351 if (!IMetricNode<Children...>::m_is_root) {
352 root["type"] = "container";
353 }
354
355 // Iterate through the descendants and recursively call get_json. The recursion will stop
356 // as soon as we reach the node-value. The obtained objects are placed with the desired
357 // name in root and return
358 for_each_tuple(IMetricNode<Children...>::m_children,
359 [this, &root](auto& child) { root[child->get_name()] = child->to_json(); });
360
361 return root;
362 }
363
364
365
366
374 std::vector<uint8_t> to_mdtp() const override {
375 std::vector<uint8_t> result;
376 result.resize(1 /* node type */ + 4 /* name length */ +
377 IMetricNode<Children...>::m_name.length() /* name */
378 + 4 /* payload size */ + 0 /* payload */,
379 0x0 /* fill by 0x0 */);
380
381 std::size_t offset = 0;
382
383 // If node type is root
384 if (IMetricNode<Children...>::m_is_root) {
385 // Add header
386 result.resize(result.size() + 1 /* version */ + 4 /* payload size */);
387 offset = 5; // After header
388 }
389
390 // Payload
391 std::vector<uint8_t> payload;
392
393 // Iterate through the descendants and recursively call to_mdtp. The recursion will stop
394 // as soon as we reach the node-value. The obtained objects are placed with the desired
395 // name in root and return
396 for_each_tuple(IMetricNode<Children...>::m_children, [this, &payload](auto& child) {
397 auto child_data = child->to_mdtp();
398 payload.insert(payload.end(), child_data.begin(), child_data.end());
399 });
400
401 // Write node type
402 write_ubyte_be(result, offset, 0);
403 ++offset;
404
405 // Write node name length
406 write_uint32_be(
407 result, offset, static_cast<uint32_t>(IMetricNode<Children...>::m_name.length()));
408 offset += 4;
409
410 // Write node name
411 std::copy(IMetricNode<Children...>::m_name.begin(),
412 IMetricNode<Children...>::m_name.end(),
413 result.data() + offset);
414 offset += IMetricNode<Children...>::m_name.length();
415
416 // Write payload size
417 write_uint32_be(result, offset, static_cast<uint32_t>(payload.size()));
418 offset += 4;
419
420 // Write payload
421 std::copy(payload.begin(), payload.end(), std::back_inserter(result));
422
423 // If node type is root, insert header
424 if (IMetricNode<Children...>::m_is_root) {
425 write_ubyte_be(result, 0, MDTP_VERSION); // write version
426 write_uint32_be(
427 result,
428 1,
429 static_cast<uint32_t>(payload.size() + 1 /* node type */ + 4 /* name length */ +
430 IMetricNode<Children...>::m_name.length() /* name */
431 + 4 /* payload size */)); // write tail size
432 }
433
434 return result;
435 }
