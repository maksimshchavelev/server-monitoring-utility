192 public:
201 MetricContainerNode(const std::string& node_name,
202 bool is_root_container,
203 Children&&... children) :
204 IMetricNode<Children...>(
205 node_name, "", "", is_root_container, std::forward<Children>(children)...) {}
206
207
208
209
214 Json::Value to_json() const override {
215 Json::Value root;
216
217 // We don't need to store the type if we are root node
218 if (!IMetricNode<Children...>::m_is_root) {
219 root["type"] = "container";
220 }
221
222 // Iterate through the descendants and recursively call get_json. The recursion will stop
223 // as soon as we reach the node-value. The obtained objects are placed with the desired
224 // name in root and return
225 for_each_tuple(IMetricNode<Children...>::m_children,
226 [this, &root](auto& child) { root[child->get_name()] = child->to_json(); });
227
228 return root;
229 }
