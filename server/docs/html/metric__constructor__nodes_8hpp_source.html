187 public:
196 MetricContainerNode(const std::string& node_name,
197 bool is_root_container,
198 Children&&... children) :
199 IMetricNode<Children...>(
200 node_name, "", "", is_root_container, std::forward<Children>(children)...) {}
201
202
203
204
209 Json::Value to_json() const override {
210 Json::Value root;
211
212 // We don't need to store the type if we are root node
213 if (!IMetricNode<Children...>::m_is_root) {
214 root["type"] = "container";
215 }
216
217 // Iterate through the descendants and recursively call get_json. The recursion will stop
218 // as soon as we reach the node-value. The obtained objects are placed with the desired
219 // name in root and return
220 for_each_tuple(IMetricNode<Children...>::m_children,
221 [this, &root](auto& child) { root[child->get_name()] = child->to_json(); });
222
223 return root;
224 }
225};
