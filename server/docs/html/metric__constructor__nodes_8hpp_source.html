273 Json::Value to_json() const override {

275
276 // We don't need to store the type if we are root node
277 if (!IMetricNode<Children...>::m_is_root) {
278 root["type"] = "container";
279 }
280
281 // Iterate through the descendants and recursively call get_json. The recursion will stop
282 // as soon as we reach the node-value. The obtained objects are placed with the desired
283 // name in root and return
284 for_each_tuple(IMetricNode<Children...>::m_children,
