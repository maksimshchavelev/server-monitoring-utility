167template <typename... Children> class MetricContainerNode : public IMetricNode<Children...> {
168 public:
177 MetricContainerNode(const std::string& node_name,
178 bool is_root_container,
179 Children&&... children) :
180 IMetricNode<Children...>(
181 node_name, "", "", is_root_container, std::forward<Children>(children)...) {}
182
183
184
185
190 Json::Value to_json() const override {
191 Json::Value root;
192
193 // We don't need to store the type if we are root node
194 if (!IMetricNode<Children...>::m_is_root) {
195 root["type"] = "container";
196 }
197
198 // Iterate through the descendants and recursively call get_json. The recursion will stop
199 // as soon as we reach the node-value. The obtained objects are placed with the desired
200 // name in root and return
201 for_each_tuple(IMetricNode<Children...>::m_children,
202 [this, &root](auto& child) { root[child->get_name()] = child->to_json(); });
203
204 return root;
205 }
206};
