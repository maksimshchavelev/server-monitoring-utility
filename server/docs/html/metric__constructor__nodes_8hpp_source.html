360 [[deprecated("Use to_mdtp() instead")]] Json::Value to_json() const override {
361 Json::Value root;
362
363 // We don't need to store the type if we are root node
364 if (!IMetricNode<Children...>::m_is_root) {
365 root["type"] = "container";
366 }
367
368#pragma GCC diagnostic push
369#pragma GCC diagnostic ignored "-Wdeprecated-declarations"
370 // Iterate through the descendants and recursively call get_json. The recursion will stop
371 // as soon as we reach the node-value. The obtained objects are placed with the desired
372 // name in root and return
373 for_each_tuple(IMetricNode<Children...>::m_children,
374 [this, &root](auto& child) { root[child->get_name()] = child->to_json(); });
375#pragma GCC diagnostic pop
376

393 std::vector<uint8_t> to_mdtp() const override {
394 // 1) Collect children payload (concatenate bytes of all children nodes).
395 // Each child already knows how to serialize itself (value or container).
396 std::vector<uint8_t> children_payload;
397 children_payload.reserve(128); // small heuristic; optional
398
399 for_each_tuple(IMetricNode<Children...>::m_children, [&](auto& child) {
400 auto bytes = child->to_mdtp();
401 children_payload.insert(children_payload.end(), bytes.begin(), bytes.end());
402 });
403
404 // 2) Root vs non-root behavior.
405 if (IMetricNode<Children...>::m_is_root) {
406 // --- Root node: only MDTP frame header + children payload ---
407 // Frame header: [version:1][payload_size:4]
408 std::vector<uint8_t> result(5, 0x00);
409
410 // write version
411 write_ubyte_be(result, 0, MDTP_VERSION);
412 // write payload size = total bytes of concatenated children
413 write_uint32_be(result, 1, static_cast<uint32_t>(children_payload.size()));
414
415 // append children payload
416 result.insert(result.end(), children_payload.begin(), children_payload.end());
417 return result;
418
419 } else {
420 // --- Regular container: container header + children payload ---
421 // Container header:
422 // [node type=0:1][name length:4][name bytes][payload size:4]
423 const uint32_t name_len =
424 static_cast<uint32_t>(IMetricNode<Children...>::m_name.size());
425
426 std::vector<uint8_t> result;
427 result.resize(1 + 4 + name_len + 4); // header without payload
428
429 size_t off = 0;
430
431 // node type = 0 (container)
432 write_ubyte_be(result, off, 0);
433 ++off;
434
435 // name length (BE)
436 write_uint32_be(result, off, name_len);
437 off += 4;
438
439 // name bytes (no terminating zero)
440 if (name_len) {
441 std::memcpy(result.data() + off, IMetricNode<Children...>::m_name.data(), name_len);
442 off += name_len;
443 }
444
445 // payload size (BE)
446 write_uint32_be(result, off, static_cast<uint32_t>(children_payload.size()));
447 off += 4;
448
449 // append children payload
450 result.insert(result.end(), children_payload.begin(), children_payload.end());
451 return result;
452 }
453 }
